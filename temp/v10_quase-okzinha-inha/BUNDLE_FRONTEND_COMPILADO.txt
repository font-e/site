function attachEvents(sectionEl) {
  const facets = document.getElementById('programacao-facets-container');
  const gridContainer = document.getElementById('programacao-grid-container');

  if (facets) {
    facets.addEventListener('click', (e) => {
      const pill = e.target.closest('.filter-pill');
      if (pill) {
        state.filter = pill.dataset.filter;
        state.activeId = null;
        state.zoomIndex = null;
        renderFacets();
        renderGrid();
      }
    });
  }

  if (gridContainer) {
    gridContainer.addEventListener('click', (e) => {
      // 1. Fechar gaveta
      if (e.target.closest('[data-action="close"]')) {
        state.activeId = null;
        state.zoomIndex = null;
        renderGrid();
        return;
      }

      // 2. Fechar Zoom
      if (e.target.closest('[data-action="close-zoom"]') || e.target.closest('[data-action="back-event"]')) {
        state.zoomIndex = null;
        renderGrid();
        return;
      }

      // 3. Abrir Zoom
      const zoomTrigger = e.target.closest('[data-action="open-zoom"]');
      if (zoomTrigger) {
        state.zoomIndex = parseInt(zoomTrigger.dataset.idx, 10) || 0;
        renderGrid();
        return;
      }

      // 4. Troca de abas internas
      const tabBtn = e.target.closest('.drawer-tab-pill');
      if (tabBtn) {
        state.activeTab = tabBtn.dataset.tab;
        renderGrid();
        return;
      }

      // 5. Navegação Próximo/Anterior
      if (e.target.closest('[data-action="prev"]')) {
        navigateEvent(-1);
        return;
      }
      if (e.target.closest('[data-action="next"]')) {
        navigateEvent(1);
        return;
      }

      // 6. Clique na linha do evento
      const header = e.target.closest('.event-header');
      if (header) {
        const article = header.closest('.programacao-event-item');
        const id = article.dataset.id;
        state.activeId = (state.activeId === id) ? null : id;
        state.activeTab = 'sobre';
        state.zoomIndex = null;
        renderGrid();

        if (state.activeId) {
          setTimeout(() => {
            const targetEl = document.querySelector(`.programacao-event-item[data-id="${id}"]`);
            if (targetEl) {
              const headerHeight = document.getElementById('logo-controller')?.offsetHeight || 64;
              const targetTop = targetEl.getBoundingClientRect().top + window.scrollY - headerHeight;
              window.scrollTo({ top: Math.max(0, targetTop), behavior: 'smooth' });
            }
          }, 30);
        }
      }
    });
  }
}

function navigateEvent(delta) {
  let list = state.events;
  if (state.filter !== 'TODA') {
    list = list.filter(e => (e.subsection || '').toUpperCase().includes(state.filter));
  }
  const currentIdx = list.findIndex(e => e.id === state.activeId);
  if (currentIdx === -1) return;

  const nextIdx = currentIdx + delta;
  if (nextIdx >= 0 && nextIdx < list.length) {
    state.activeId = list[nextIdx].id;
    state.zoomIndex = null;
    state.activeTab = 'sobre';
    renderGrid();
  }
}

import { generateCrossMatrix } from './vectors.js';

export function initCanvasEngine() {
  const canvasNormal = document.getElementById('canvas-normal');
  const canvasHighlight = document.getElementById('canvas-highlight');
  if (!canvasNormal || !canvasHighlight) return;

  const ctxNormal = canvasNormal.getContext('2d', { alpha: true });
  const ctxHighlight = canvasHighlight.getContext('2d', { alpha: true });
  if (!ctxNormal || !ctxHighlight) return;

  const CONFIG = {
    strokeWidth: 3.5,
    colorNormal: '#ffffff',
    colorHighlight: '#ffffff'
  };

  let cells = [];
  let canvasWidth = 0;
  let canvasHeight = 0;
  let isCompleteMode = false;
  let activeHighlightSet = new Set(['f', 'n']); // Destaque padrão da primeira seção (Programação)

  /* Dicionário Oficial de Destaques por Seção */
  const sectionHighlightsMap = {
    'sec-programacao': ['f', 'n'],
    'sec-residencia':  ['e', 't'],
    'sec-atelies':     ['o', 'n'],
    'sec-arquivo':     ['f', 't']
  };

  function updateGrid() {
    const isMobile = canvasWidth < 600;
    const matrix = generateCrossMatrix({
      width: canvasWidth,
      height: canvasHeight,
      isMobile
    });
    cells = matrix.cells;
  }

  function drawPattern() {
    ctxNormal.clearRect(0, 0, canvasWidth, canvasHeight);
    ctxHighlight.clearRect(0, 0, canvasWidth, canvasHeight);

    ctxNormal.lineCap = 'square';
    ctxNormal.lineJoin = 'miter';
    ctxHighlight.lineCap = 'square';
    ctxHighlight.lineJoin = 'miter';

    for (let i = 0; i < cells.length; i++) {
      const cell = cells[i];
      const isHighlight = activeHighlightSet.has(cell.letter);
      const ctx = isHighlight ? ctxHighlight : ctxNormal;
      const path = isCompleteMode ? cell.completePath : cell.path;

      if (!path) continue;

      ctx.save();
      ctx.translate(cell.cx, cell.cy);
      ctx.scale(cell.scale, cell.scale);
      ctx.translate(-530, -530); // 1060 / 2 para ancoragem no centro exato da célula

      ctx.lineWidth = CONFIG.strokeWidth / cell.scale;
      ctx.strokeStyle = isHighlight ? CONFIG.colorHighlight : CONFIG.colorNormal;
      ctx.stroke(path);

      ctx.restore();
    }
  }

  function resizeCanvases() {
    const dpr = window.devicePixelRatio || 1;
    canvasWidth = window.innerWidth;
    canvasHeight = window.innerHeight;

    canvasNormal.width = Math.floor(canvasWidth * dpr);
    canvasNormal.height = Math.floor(canvasHeight * dpr);
    ctxNormal.setTransform(1, 0, 0, 1, 0, 0);
    ctxNormal.scale(dpr, dpr);

    canvasHighlight.width = Math.floor(canvasWidth * dpr);
    canvasHighlight.height = Math.floor(canvasHeight * dpr);
    ctxHighlight.setTransform(1, 0, 0, 1, 0, 0);
    ctxHighlight.scale(dpr, dpr);

    updateGrid();
    drawPattern();
  }

  function initObserver() {
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const sectionId = entry.target.id;
          const lettersToHighlight = sectionHighlightsMap[sectionId];
          if (lettersToHighlight) {
            activeHighlightSet = new Set(lettersToHighlight);
            drawPattern();
          }
        }
      });
    }, { threshold: 0.35 });

    document.querySelectorAll('.section-block').forEach(section => {
      observer.observe(section);
    });
  }

  resizeCanvases();
  window.addEventListener('resize', resizeCanvases, { passive: true });
  initObserver();

  /* Alternância de Modo Completo via Tecla 'P' */
  window.addEventListener('keydown', (e) => {
    if (e.target && (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA' || e.target.isContentEditable)) return;
    if (e.key === 'p' || e.key === 'P') {
      isCompleteMode = !isCompleteMode;
      drawPattern();
    }
  });

  // Expõe a função de redesenho para integração com o futuro laboratório sandbox
  window._fonteRedrawCanvas = () => {
    updateGrid();
    drawPattern();
  };
}

function attachEvents(sectionEl) {

  const photosGrid = sectionEl.querySelector('.photos-grid');
  
  // Hover Bidirecional Otimizado e Event Delegation
  const syncHover = (id, isHovering) => {
    const btn = sectionEl.querySelector(`.artist-name-btn[data-id="${id}"]`);
    const cell = sectionEl.querySelector(`.photo-cell[data-id="${id}"]`);
    if (btn) btn.classList.toggle('is-hovered', isHovering);
    if (cell) cell.classList.toggle('is-hovered', isHovering);
  };

  sidebar.addEventListener('mouseover', (e) => {
    const btn = e.target.closest('.artist-name-btn');
    if (btn && !state.activeArtistId) syncHover(btn.dataset.id, true);
  });
  sidebar.addEventListener('mouseout', (e) => {
    const btn = e.target.closest('.artist-name-btn');
    if (btn && !state.activeArtistId) syncHover(btn.dataset.id, false);
  });
  
  photosGrid.addEventListener('mouseover', (e) => {
    const cell = e.target.closest('.photo-cell');
    if (cell && !state.activeArtistId) syncHover(cell.dataset.id, true);
  });
  photosGrid.addEventListener('mouseout', (e) => {
    const cell = e.target.closest('.photo-cell');
    if (cell && !state.activeArtistId) syncHover(cell.dataset.id, false);
  });

  // Abertura de Perfil
  const handleSelect = (e) => {
    const el = e.target.closest('.artist-name-btn') || e.target.closest('.photo-cell');
    if (el) openArtistProfile(el.dataset.id, sectionEl);
  };
  
  sidebar.addEventListener('click', handleSelect);
  photosGrid.addEventListener('click', handleSelect);
}

function toList(val) {
  if (Array.isArray(val)) return val.filter(Boolean);
  if (typeof val === 'string' && val.trim().length > 0) return [val.trim()];

// Deriva Pessoas e Textos a partir dos Eventos (Dispensa o pesado PESSOAS_CONSOLIDADO)
function buildDerivedDatasets(events) {
  state.data.EVENTOS = events.sort((a, b) => (b.ano || 0) - (a.ano || 0));
  
  const personsMap = new Map();
  const textsList = [];

  events.forEach(evt => {
    const allPeople = [...toList(evt.artistas), ...toList(evt.curadoria)];
    allPeople.forEach(name => {
      if (!personsMap.has(name)) {
        personsMap.set(name, { id: name.replace(/\s+/g, '-').toLowerCase(), nome: name, acoes: [] });
      }
      personsMap.get(name).acoes.push(evt);
    });

    if (evt.textos && Array.isArray(evt.textos)) {
      evt.textos.forEach((txt, idx) => {
        textsList.push({
          id: `${evt.id}-txt-${idx}`,
          eventoId: evt.id,
          eventoTitle: evt.title,
          ano: evt.ano,
          ...txt
        });
      });
    }
  });

  state.data.PESSOAS = Array.from(personsMap.values()).sort((a, b) => b.acoes.length - a.acoes.length);
  state.data.TEXTOS = textsList;
}

/* 3. GERADOR DE MATRIZ TIPOGRÁFICA CRUZADA (FONTE VERTICAL & HORIZONTAL) */
export function generateCrossMatrix({ width, height, isMobile }) {
  const cols = isMobile ? 6 : 12;
  const colWidth = width / cols;
  const colHeight = colWidth; // Proporção quadrada estrita
  const rows = Math.ceil(height / colHeight) + 1;
  const usableSize = colWidth * 0.76;
  const scale = usableSize / VIEWBOX_SIZE;

  // Contadores independentes de versão (1 a 5) para cada letra
  const versionCounters = {
    f: 1,
    o: 1,
    n: 1,
    t: 1,
    e: 1
  };

  const cells = [];

  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      // 1. Identificação da letra com leitura cruzada: (r + c) % 5
      const letter = LETTERS[(r + c) % 5];

      // 2. Extração da versão atual da letra e rotação circular para a próxima
      const version = versionCounters[letter];
      versionCounters[letter] = (version % 5) + 1;

      const cx = c * colWidth + colWidth / 2;
      const cy = r * colHeight + colHeight / 2;

      cells.push({
        r,
        c,
        cx,
        cy,
        letter,
        version,
        scale,
        key: `${letter}${version}`,
        path: FRAGMENT_PATHS[`${letter}${version}`],
        completePath: COMPLETE_PATHS[letter]
      });
    }
  }

  return { cells, cols, rows, colWidth, colHeight };
}
